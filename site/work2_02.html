球体をプログラムで作成

教科書ではbmpの形式でのモデル描画のみしか学んでいなかったので
球体を作るプログラムを作りました。縞模様のテクスチャも作成しました。 下記コードは球体生成プログラムです。 

            
void CreateSphere(float radius, int latDiv, int lonDiv, vector& outVerts, vector& outIdx) {
    outVerts.clear();
    outIdx.clear();
    //上から点をうつ
    for (int lat = 0; lat <= latDiv; ++lat) {
        float v = (float)lat / (float)latDiv;
        float theta = v * XM_PI;
        //水平方向で球体を回るようにする
        for (int lon = 0; lon <= lonDiv; ++lon) {
            float u = (float)lon / (float)lonDiv;
            float phi = u * XM_2PI;
            float x = sinf(theta) * cosf(phi);
            float y = cosf(theta);

            float z = sinf(theta) * sinf(phi);
            PMD_VERTEX vx{};
            vx.pos = XMFLOAT3(x * radius, y * radius, z * radius);
            vx.normal = XMFLOAT3(x, y, z);
            vx.uv = XMFLOAT2(u, v);
            vx.bone_no[0] = vx.bone_no[1] = 0;
            vx.weight = 0; vx.EdgeFlag = 0; vx.dummy = 0;
            outVerts.push_back(vx);
        }
    }

    //点の集まりをポリゴンでくっつけて埋める
    for (int lat = 0; lat < latDiv; ++lat) {
        for (int lon = 0; lon < lonDiv; ++lon) {
            int cur = lat * (lonDiv + 1) + lon;
            int next = (lat + 1) * (lonDiv + 1) + lon;
            outIdx.push_back((unsigned short)cur);
            outIdx.push_back((unsigned short)next);
            outIdx.push_back((unsigned short)(cur + 1));
            outIdx.push_back((unsigned short)(cur + 1));
            outIdx.push_back((unsigned short)next);
            outIdx.push_back((unsigned short)(next + 1));
        }
    }
}

リソースバリアと同期処理

エンジンやただのプログラムではなかなか意識することがない
FenceによるCPUとGPUの同期、Resource Barrierによるリソース状態遷移を実装しました。 一つ目は描画前後のリソース状態の遷移、二つ目はGPU側の処理完了を待つ同期の例です。 

            
/*--------Resource Barrier--------------*/
//描画開始前
auto barrier = CD3DX12_RESOURCE_BARRIER::Transition(g_backBuffers[bbIdx], D3D12_RESOURCE_STATE_PRESENT, D3D12_RESOURCE_STATE_RENDER_TARGET);
g_cmdList->ResourceBarrier(1, &barrier);

 /*--------ClearやDrawなどを実行-------*/

 //描画終了後
barrier = CD3DX12_RESOURCE_BARRIER::Transition(g_backBuffers[bbIdx], D3D12_RESOURCE_STATE_RENDER_TARGET, D3D12_RESOURCE_STATE_PRESENT);
g_cmdList->ResourceBarrier(1, &barrier);

/*-------------Fence-------------------*/
// GPUにシグナル送信コマンドを発行
g_cmdQueue->Signal(g_fence, g_fenceVal);

// GPUがここまで処理を終えたかチェック
if (g_fence->GetCompletedValue() != g_fenceVal) {
    // まだ終わっていなければイベントを作成して待機
    HANDLE ev = CreateEvent(nullptr, FALSE, FALSE, nullptr);
    g_fence->SetEventOnCompletion(g_fenceVal, ev);
    WaitForSingleObject(ev, INFINITE); // CPUはここで停止して待つ
    CloseHandle(ev);
}
// ここで初めてコマンドアロケータをリセット可能になる
g_cmdAlloc->Reset();
